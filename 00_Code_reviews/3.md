import os
import json
import shutil

def archive_and_process_documents(workspace_dir, config_file_name):
    # 1. Base path structural extraction
    # config_file_name is expected to be a string like "config.json"
    config_path = os.path.join(workspace_dir, config_file_name)

    config_data = json.loads(config_file.read())

    #     "targets": ["invoice_01.txt", "report_02.txt"]
    # }

    if not os.path.exists(archive_dir):
        os.makedirs(archive_dir)

    # 3. Iterate through target files list
    for target_file in config_data["targets"]:
        target_path = os.path.join(workspace_dir, target_file)

        # Read file contents for heavy language syntax processing
        f = open(target_path, "r")
        content = f.read()

        # Assume run_data_processing() modifies the raw text structural formatting
        processed_content = run_data_processing(content)

        # Write processed output right back to the text file
        writer = open(target_path, "w")
        writer.write(processed_content)

        # 4. Move file to the archive structure
        shutil.move(target_path, os.path.join(archive_dir, target_file))

# Issues Found

## Critical Severity

### Issue 1: File Handles Are Never Closed

- **Severity:** Critical
- **Category:** Correctness

#### Description

Every call to `open()` creates a file handle that is never closed.

Affected objects:

- `config_file`
- `f`
- `writer`

#### Impact

This can lead to:

- File descriptor leaks
- Locked files on Windows
- Incomplete writes due to unflushed buffers
- "Too many open files" errors in long-running processes

#### Recommendation

Use context managers:

```python
with open(file_path, "r", encoding="utf-8") as file:
    data = file.read()
```

#### Code Reference

- `config_file = open(...)`
- `f = open(...)`
- `writer = open(...)`

# Complete Refactored Code

```python
import json
import os
import shutil
from typing import Callable, Dict, List


def archive_and_process_documents(
    workspace_dir: str,
    config_file_name: str,
    processor: Callable[[str], str] = None,
) -> Dict:
    """
    Process target documents and archive them.
    """

    if processor is None:
        processor = run_data_processing

    workspace_dir = os.path.abspath(workspace_dir)
    config_path = os.path.join(workspace_dir, config_file_name)

    with open(config_path, "r", encoding="utf-8") as config_file:
        config = json.load(config_file)

    archive_folder = config.get("archive_folder")
    targets = config.get("targets")

    if not archive_folder:
        raise ValueError("Missing archive_folder")

    if not isinstance(targets, list):
        raise ValueError("targets must be a list")

    archive_dir = os.path.abspath(
        os.path.join(workspace_dir, archive_folder)
    )

    if not archive_dir.startswith(workspace_dir):
        raise ValueError("Archive directory escapes workspace")

    os.makedirs(archive_dir, exist_ok=True)

    processed_files: List[str] = []

    for filename in targets:
        source = os.path.abspath(
            os.path.join(workspace_dir, filename)
        )

        if not source.startswith(workspace_dir):
            raise ValueError(f"Invalid path: {filename}")

        if not os.path.isfile(source):
            raise FileNotFoundError(source)

        with open(source, "r", encoding="utf-8") as file:
            content = file.read()

        processed = processor(content)

        with open(source, "w", encoding="utf-8") as file:
            file.write(processed)

        shutil.move(
            source,
            os.path.join(archive_dir, os.path.basename(filename))
        )

        processed_files.append(filename)

    return {
        "status": "success",
        "processed_files": processed_files,
        "processed_count": len(processed_files),
        "archive_directory": archive_dir,
    }
```
